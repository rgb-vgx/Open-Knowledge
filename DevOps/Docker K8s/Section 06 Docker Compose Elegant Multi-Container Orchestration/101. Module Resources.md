```
│   ├── Dockerfile

│   └── backend.env

```

## Docker Compose File Template

```yaml
version: "3.8"

services:
  # Service 1: Database
  mongodb:
    image: "mongo"
    volumes:
      - data:/data/db
    environment:
      MONGO_INITDB_ROOT_USERNAME: max
      MONGO_INITDB_ROOT_PASSWORD: secret

  # Service 2: Backend API
  backend:
    build: ./backend
    ports:
      - "80:80"
    volumes:
      - logs:/app/logs
      - ./backend:/app
      - /app/node_modules
    env_file:
      - ./env/backend.env
    depends_on:
      - mongodb

  # Service 3: Frontend
  frontend:
    build: ./frontend
    ports:
      - "3000:3000"
    volumes:
      - ./frontend/src:/app/src
    stdin_open: true
    tty: true
    depends_on:
      - backend

volumes:
  data:
  logs:
```

## Lệnh Thường Dùng

```bash
# Khởi chạy toàn bộ
docker compose up

# Chạy nền
docker compose up -d

# Force rebuild
docker compose up --build

# Dừng services
docker compose down

# Dừng và xóa volumes
docker compose down -v

# Chỉ build images
docker compose build

# Xem logs
docker compose logs
docker compose logs backend

# Xem services đang chạy
docker compose ps
```
